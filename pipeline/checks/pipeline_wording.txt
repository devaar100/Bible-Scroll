# Wording that leaks the generation pipeline into a study. Same format as banned.txt (one phrase per line,
# case-insensitive, whole words; "re:" lines are regular expressions). Unlike banned.txt, a match is never
# excused by a nearby "not"/"no": these words shouldn't appear at all.
re: \bcross[- ]?references?\b
# "supplied" as the prompt's word ("the words supplied"); not "supplied with" in a narrative ("were supplied with
# food", Psalm 78:29). Checker 16 / devotionals-8.
re: \bsupplied\b(?!\s+with\b)

# Leaked instructions (decision 25). "the/these instructions" only when it points at the writer's own input
# ("Follow these instructions", "per the instructions", "the instructions given/above", "the instructions ask me");
# instructions given by someone in the text pass ("The instructions that follow tell them to take a staff only",
# Mark 6:8). Checker 15 / devotionals-7.
re: \b(?:follow|following|per|according to) these instructions\b
re: \b(?:per|according to) the instructions\b
re: \b(?:the|these) instructions (?:given|provided|above|below)\b
re: \b(?:the|these) instructions (?:say|said|ask|asked) (?:me|us|for)\b
word range
# "word count" as the prompt's term; not when a quoted word follows (the word count “number”). Checker 14.
re: (?<!\w)word\s+count(?!\w)(?!\s*["“‘'])

# "the verses given" and the like talk about the writer's input (checker 17 / devotionals-9). Ordinary uses pass:
# "words given in advance", "the words given to them", "with no words given".
re: \b(?:the|these) (?:verses?|lines|text|passages?) (?:given|provided)\b(?!\s+(?:to|by|for|in advance)\b)
re: \b(?:verses|lines|words) given (?:with|around|alongside) (?:it|this|the)\b
re: \bthe words given(?: here)?(?=[,.:;]|\s+here\b)
